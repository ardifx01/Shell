	<script>
		$(document).ready(function() {		
			$('#submitlogin').click( function() {							
				$('#submitload').html('<i class="fa fa-circle-o-notch fa-spin"></i>');				
				if($('#userlogin').val()!='' && $('#userpass').val()!='') {
					$.ajax({
						url: 'users/ext_login',
						type: "POST",
						headers: { "cache-control": "no-cache" },
						async: true,
						dataType: "json",
						data: {
							username: $('#userlogin').val(),
							password: $('#userpass').val()
						},
						success: function(jsonData) {
							if(jsonData.success=='OK') window.location.assign(jsonData.redirect);
							else {
								alert('Username atau password salah!');	
								$('#submitload').html('Login');
							}
						}
					});					
					return false;
				} else $('#submitload').html('Login');
			});
		});
	</script>
